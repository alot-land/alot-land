create role authenticated nologin;
create role anon nologin;
create schema auth;
create table auth.users (id uuid primary key, email text);
create function auth.uid() returns uuid language sql stable as $$ select nullif(current_setting('audit.uid',true),'')::uuid $$;
create function auth.jwt() returns jsonb language sql stable as $$ select jsonb_build_object('email',current_setting('audit.email',true)) $$;
create schema storage;
create table storage.buckets (id text primary key, name text, public boolean default false);
grant usage on schema public,auth to authenticated,anon;
